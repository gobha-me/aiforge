cmake_minimum_required(VERSION 3.28)

project(dependency_probe LANGUAGES C CXX)

if (NOT AIFORGE_SOURCE_DIR)
  message(FATAL_ERROR "AIFORGE_SOURCE_DIR is required")
endif ()

if (PROBE_PREEXISTING_TARGETS)
  if (NOT IS_DIRECTORY "${PROBE_TERMFORGE_INCLUDE_DIR}")
    message(FATAL_ERROR
      "PROBE_TERMFORGE_INCLUDE_DIR is required for pre-existing targets")
  endif ()

  add_library(probe_termforge INTERFACE)
  target_include_directories(probe_termforge INTERFACE
    "${PROBE_TERMFORGE_INCLUDE_DIR}")
  add_library(termforge::lib ALIAS probe_termforge)
  # Reuse an actual installed target; an empty placeholder cannot prove the
  # public HTTP/header contract or the c-ares link closure.
  find_package(venice-cpp 0.29.18 CONFIG REQUIRED)
  add_library(probe_rasterforge INTERFACE)
  add_library(rasterforge::lib ALIAS probe_rasterforge)
  add_library(SQLite::SQLite3 INTERFACE IMPORTED)
  if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(dbus-1 INTERFACE IMPORTED)
    set(DBUS_BUILD_TESTS ON CACHE BOOL "sentinel" FORCE)
    include("${AIFORGE_SOURCE_DIR}/cmake/deps/dbus1.cmake")
    if (NOT DBUS_BUILD_TESTS)
      message(FATAL_ERROR "A pre-existing libdbus target must prevent option mutation")
    endif ()
  endif ()

  add_library(yaml-cpp::yaml-cpp INTERFACE IMPORTED)
  set(YAML_CPP_BUILD_TOOLS ON CACHE BOOL "sentinel" FORCE)

  set(termforge_TESTS ON CACHE BOOL "sentinel" FORCE)
  set(venice-cpp_TESTS ON CACHE BOOL "sentinel" FORCE)
  set(rasterforge_TESTS ON CACHE BOOL "sentinel" FORCE)

  include("${AIFORGE_SOURCE_DIR}/cmake/deps/termforge.cmake")
  include("${AIFORGE_SOURCE_DIR}/cmake/deps/venice_cpp.cmake")
  include("${AIFORGE_SOURCE_DIR}/cmake/deps/rasterforge.cmake")
  include("${AIFORGE_SOURCE_DIR}/cmake/deps/sqlite3.cmake")
  include("${AIFORGE_SOURCE_DIR}/cmake/deps/yaml_cpp.cmake")

  if (NOT termforge_TESTS OR NOT venice-cpp_TESTS OR NOT rasterforge_TESTS)
    message(FATAL_ERROR "A pre-existing canonical target must prevent option mutation")
  endif ()
  if (NOT TARGET SQLite3::SQLite3)
    message(FATAL_ERROR "The legacy SQLite target was not normalized")
  endif ()
  if (NOT YAML_CPP_BUILD_TOOLS)
    message(FATAL_ERROR "Pre-existing yaml-cpp target must prevent option mutation")
  endif ()
  return()
endif ()

if (NOT PROBE_DEPENDENCY MATCHES "^(termforge|venice_cpp|rasterforge|sqlite3|dbus1|yaml_cpp|systemd_journal)$")
  message(FATAL_ERROR
    "PROBE_DEPENDENCY must be termforge, venice_cpp, rasterforge, sqlite3, dbus1, yaml_cpp, or systemd_journal")
endif ()

if (PROBE_DEPENDENCY STREQUAL "venice_cpp")
  # A normal binding shadows a different cache preference. Neither may leak
  # from the acquisition-only target-promotion scope in the production recipe.
  set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL ON CACHE BOOL "caller cache sentinel" FORCE)
  set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL OFF)
endif ()

if (PROBE_EXPECT_EMBEDDED)
  if (PROBE_DEPENDENCY STREQUAL "dbus1")
    foreach(option IN ITEMS DBUS_BUILD_TESTS DBUS_ENABLE_INTRUSIVE_TESTS
        DBUS_ENABLE_VERBOSE_MODE DBUS_BUILD_X11 DBUS_WITH_GLIB
        DBUS_ENABLE_DOXYGEN_DOCS DBUS_ENABLE_XML_DOCS DBUS_ENABLE_MODULAR_TESTS
        DBUS_DISABLE_CHECKS)
      set(${option} ON CACHE BOOL "dependency-probe sentinel" FORCE)
    endforeach()
    set(ENABLE_QT_HELP ON CACHE STRING "dependency-probe sentinel" FORCE)
    set(ENABLE_SYSTEMD ON CACHE STRING "dependency-probe sentinel" FORCE)
  endif ()
  foreach(option IN ITEMS BUILD_TESTS BUILD_TOOLS BUILD_CONTRIB INSTALL FORMAT_SOURCE)
    set(YAML_CPP_${option} ON CACHE BOOL "dependency-probe sentinel" FORCE)
  endforeach()
  set(YAML_BUILD_SHARED_LIBS ON CACHE BOOL "dependency-probe sentinel" FORCE)
  set(YAML_CPP_DISABLE_UNINSTALL OFF CACHE BOOL "dependency-probe sentinel" FORCE)
  set(YAML_ENABLE_PIC OFF CACHE BOOL "dependency-probe sentinel" FORCE)
  foreach(option IN ITEMS TESTS EXAMPLES BIN TOOLS BENCH INSTALL)
    set(termforge_${option} ON CACHE BOOL "dependency-probe sentinel" FORCE)
  endforeach()
  foreach(option IN ITEMS BUILD_BIN TESTS INSTALL)
    set(venice-cpp_${option} ON CACHE BOOL "dependency-probe sentinel" FORCE)
  endforeach()
  set(rasterforge_BUILD_LIB OFF CACHE BOOL "dependency-probe sentinel" FORCE)
  foreach(option IN ITEMS BUILD_BIN TESTS FUZZERS BENCHMARKS INSTALL)
    set(rasterforge_${option} ON CACHE BOOL "dependency-probe sentinel" FORCE)
  endforeach()
endif ()

if (PROBE_DEPENDENCY STREQUAL "systemd_journal" AND PROBE_SYSTEMD_PREEXISTING)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PROBE_SYSTEMD REQUIRED IMPORTED_TARGET "libsystemd>=233")
  add_library(SystemdJournal::libsystemd INTERFACE IMPORTED)
  set_target_properties(SystemdJournal::libsystemd PROPERTIES
    INTERFACE_LINK_LIBRARIES PkgConfig::PROBE_SYSTEMD
    SystemdJournal_VERSION "${PROBE_SYSTEMD_VERSION}")
endif ()

if (PROBE_DEPENDENCY STREQUAL "venice_cpp" AND PROBE_VENICE_PREEXISTING)
  if (PROBE_VENICE_DIRECT_TARGET)
    # Import the actual exported targets without executing Venice's config
    # checker. This models a caller-provided canonical target honestly.
    find_package(OpenSSL 3 REQUIRED)
    find_package(Threads REQUIRED)
    find_package(c-ares 1.34.0 CONFIG REQUIRED)
    find_package(nlohmann_json 3.11 CONFIG REQUIRED)
    if (PROBE_VENICE_BAD STREQUAL "compiled")
      add_library(httplib::httplib STATIC IMPORTED)
      set_property(TARGET httplib::httplib PROPERTY IMPORTED_LOCATION "${CMAKE_CURRENT_BINARY_DIR}/unused.a")
    else ()
      find_package(httplib 0.51...<0.52 CONFIG REQUIRED COMPONENTS OpenSSL)
    endif ()
    if (NOT EXISTS "${PROBE_VENICE_EXPORTS}")
      message(FATAL_ERROR "Actual Venice target export fixture is missing")
    endif ()
    include("${PROBE_VENICE_EXPORTS}")
  else ()
    find_package(venice-cpp 0.29.18 CONFIG REQUIRED)
  endif ()
  get_target_property(probe_venice_target venice-cpp::lib ALIASED_TARGET)
  if (NOT probe_venice_target)
    set(probe_venice_target venice-cpp::lib)
  endif ()
  get_target_property(probe_http_target httplib::httplib ALIASED_TARGET)
  if (NOT probe_http_target)
    set(probe_http_target httplib::httplib)
  endif ()
  if (PROBE_VENICE_BAD STREQUAL "marker")
    set_property(TARGET ${probe_venice_target} PROPERTY VENICE_CPP_HTTP_TRANSPORT_CONTRACT "")
  elseif (PROBE_VENICE_BAD STREQUAL "headers")
    target_compile_definitions(${probe_http_target} INTERFACE CPPHTTPLIB_HEADER_MAX_COUNT=99)
  elseif (PROBE_VENICE_BAD STREQUAL "header_length")
    target_compile_definitions(${probe_http_target} INTERFACE CPPHTTPLIB_HEADER_MAX_LENGTH=8191)
  elseif (PROBE_VENICE_BAD STREQUAL "line_length")
    target_compile_definitions(${probe_http_target} INTERFACE CPPHTTPLIB_MAX_LINE_LENGTH=32767)
  elseif (PROBE_VENICE_BAD STREQUAL "resolver")
    target_compile_definitions(${probe_http_target} INTERFACE CPPHTTPLIB_USE_NON_BLOCKING_GETADDRINFO)
  elseif (PROBE_VENICE_BAD STREQUAL "exceptions")
    target_compile_definitions(${probe_http_target} INTERFACE CPPHTTPLIB_NO_EXCEPTIONS)
  elseif (PROBE_VENICE_BAD STREQUAL "tls")
    target_compile_definitions(${probe_http_target} INTERFACE CPPHTTPLIB_WOLFSSL_SUPPORT)
  elseif (PROBE_VENICE_BAD STREQUAL "tls_off")
    target_compile_options(${probe_http_target} INTERFACE -UCPPHTTPLIB_OPENSSL_SUPPORT)
  elseif (PROBE_VENICE_BAD STREQUAL "links")
    set_property(TARGET ${probe_http_target} PROPERTY INTERFACE_LINK_LIBRARIES "")
  elseif (PROBE_VENICE_BAD STREQUAL "include")
    if (NOT IS_DIRECTORY "${PROBE_VENICE_HTTP_INCLUDE}")
      message(FATAL_ERROR "Changed HTTP include fixture is missing")
    endif ()
    target_include_directories(${probe_http_target} BEFORE INTERFACE "${PROBE_VENICE_HTTP_INCLUDE}")
  elseif (PROBE_VENICE_BAD STREQUAL "ambient")
    if (NOT IS_DIRECTORY "${PROBE_VENICE_AMBIENT_INCLUDE}")
      message(FATAL_ERROR "Ambient Venice include fixture is missing")
    endif ()
    target_include_directories(${probe_venice_target} BEFORE INTERFACE "${PROBE_VENICE_AMBIENT_INCLUDE}")
  elseif (PROBE_VENICE_BAD AND NOT PROBE_VENICE_BAD STREQUAL "compiled")
    message(FATAL_ERROR "Unknown Venice failure fixture")
  endif ()
  # Real imported aliases, including prefix-related names in expressions.
  add_library(probe_alias_one INTERFACE IMPORTED)
  add_library(probe_alias_two INTERFACE IMPORTED)
  add_library(probe::transport ALIAS probe_alias_one)
  add_library(probe::transport_more ALIAS probe_alias_two)
  target_link_libraries(${probe_http_target} INTERFACE
    "$<LINK_ONLY:probe::transport>" "$<LINK_ONLY:probe::transport_more>")
  set(venice-cpp_TESTS ON CACHE BOOL "preexisting caller sentinel" FORCE)
endif ()

set(${PROJECT_NAME}_DEPS "${PROBE_DEPENDENCY}")
include("${AIFORGE_SOURCE_DIR}/cmake/dependencies.cmake")

if (PROBE_DEPENDENCY STREQUAL "venice_cpp")
  if (NOT CMAKE_FIND_PACKAGE_TARGETS_GLOBAL STREQUAL "OFF" OR
      NOT "$CACHE{CMAKE_FIND_PACKAGE_TARGETS_GLOBAL}" STREQUAL "ON")
    message(FATAL_ERROR "Venice acquisition changed the caller target-promotion preference")
  endif ()
  if (PROBE_VENICE_PREEXISTING AND NOT venice-cpp_TESTS)
    message(FATAL_ERROR "Preexisting Venice target changed caller options")
  endif ()
  if (PROBE_EXPECT_IMPORTED)
    get_target_property(venice_target venice-cpp::lib ALIASED_TARGET)
    if (NOT venice_target)
      set(venice_target venice-cpp::lib)
    endif ()
    get_target_property(venice_imported ${venice_target} IMPORTED)
    if (NOT venice_imported)
      message(FATAL_ERROR "Installed Venice probe unexpectedly used a source fallback")
    endif ()
  endif ()
endif ()

if (PROBE_DEPENDENCY STREQUAL "dbus1" AND PROBE_EXPECT_IMPORTED)
  get_target_property(dbus_target dbus-1 ALIASED_TARGET)
  if (NOT dbus_target)
    set(dbus_target dbus-1)
  endif ()
  get_target_property(dbus_imported ${dbus_target} IMPORTED)
  if (NOT dbus_imported)
    message(FATAL_ERROR "Installed DBus1 probe unexpectedly used a source fallback")
  endif ()
endif ()

if (PROBE_DEPENDENCY STREQUAL "yaml_cpp" AND PROBE_EXPECT_IMPORTED)
  get_target_property(yaml_target yaml-cpp::yaml-cpp ALIASED_TARGET)
  if (NOT yaml_target)
    set(yaml_target yaml-cpp::yaml-cpp)
  endif ()
  get_target_property(yaml_imported ${yaml_target} IMPORTED)
  if (NOT yaml_imported)
    message(FATAL_ERROR "Installed yaml-cpp probe unexpectedly used a source fallback")
  endif ()
endif ()

if (PROBE_EXPECT_EMBEDDED)
  if (PROBE_DEPENDENCY STREQUAL "termforge")
    foreach(option IN ITEMS TESTS EXAMPLES BIN TOOLS BENCH INSTALL)
      if (termforge_${option})
        message(FATAL_ERROR "Embedded termforge option ${option} leaked on")
      endif ()
    endforeach()
  elseif (PROBE_DEPENDENCY STREQUAL "venice_cpp")
    foreach(option IN ITEMS BUILD_BIN TESTS INSTALL)
      if (venice-cpp_${option})
        message(FATAL_ERROR "Embedded venice-cpp option ${option} leaked on")
      endif ()
    endforeach()
  elseif (PROBE_DEPENDENCY STREQUAL "rasterforge")
    if (NOT rasterforge_BUILD_LIB)
      message(FATAL_ERROR "Embedded RasterForge library was disabled")
    endif ()
    foreach(option IN ITEMS BUILD_BIN TESTS FUZZERS BENCHMARKS INSTALL)
      if (rasterforge_${option})
        message(FATAL_ERROR "Embedded RasterForge option ${option} leaked on")
      endif ()
    endforeach()
  elseif (PROBE_DEPENDENCY STREQUAL "dbus1")
    foreach(option IN ITEMS DBUS_BUILD_TESTS DBUS_ENABLE_INTRUSIVE_TESTS
        DBUS_ENABLE_VERBOSE_MODE DBUS_BUILD_X11 DBUS_WITH_GLIB
        DBUS_ENABLE_DOXYGEN_DOCS DBUS_ENABLE_XML_DOCS DBUS_ENABLE_MODULAR_TESTS
        DBUS_DISABLE_CHECKS ENABLE_QT_HELP ENABLE_SYSTEMD)
      if (${option})
        message(FATAL_ERROR "Embedded libdbus option ${option} leaked on")
      endif ()
    endforeach()

  elseif (PROBE_DEPENDENCY STREQUAL "yaml_cpp")
    foreach(option IN ITEMS BUILD_TESTS BUILD_TOOLS BUILD_CONTRIB INSTALL FORMAT_SOURCE)
      if (YAML_CPP_${option})
        message(FATAL_ERROR "Embedded yaml-cpp option ${option} leaked on")
      endif ()
    endforeach()
    if (YAML_BUILD_SHARED_LIBS OR NOT YAML_CPP_DISABLE_UNINSTALL OR NOT YAML_ENABLE_PIC)
      message(FATAL_ERROR "Embedded yaml-cpp linkage/install options were not bounded")
    endif ()
  endif ()
endif ()

add_executable(dependency_probe main.cpp)
target_compile_features(dependency_probe PRIVATE cxx_std_23)

if (PROBE_DEPENDENCY STREQUAL "termforge")
  target_compile_definitions(dependency_probe PRIVATE PROBE_TERMFORGE)
  target_link_libraries(dependency_probe PRIVATE termforge::lib)
elseif (PROBE_DEPENDENCY STREQUAL "venice_cpp")
  target_compile_definitions(dependency_probe PRIVATE PROBE_VENICE_CPP)
  target_link_libraries(dependency_probe PRIVATE venice-cpp::lib)
elseif (PROBE_DEPENDENCY STREQUAL "rasterforge")
  target_compile_definitions(dependency_probe PRIVATE PROBE_RASTERFORGE)
  target_link_libraries(dependency_probe PRIVATE rasterforge::lib)
elseif (PROBE_DEPENDENCY STREQUAL "dbus1")
  target_compile_definitions(dependency_probe PRIVATE PROBE_DBUS1)
  target_link_libraries(dependency_probe PRIVATE dbus-1)

elseif (PROBE_DEPENDENCY STREQUAL "systemd_journal")
  target_compile_definitions(dependency_probe PRIVATE PROBE_SYSTEMD_JOURNAL)
  target_link_libraries(dependency_probe PRIVATE SystemdJournal::libsystemd)
elseif (PROBE_DEPENDENCY STREQUAL "yaml_cpp")
  target_compile_definitions(dependency_probe PRIVATE PROBE_YAML_CPP)
  target_link_libraries(dependency_probe PRIVATE yaml-cpp::yaml-cpp)
else ()
  target_compile_definitions(dependency_probe PRIVATE PROBE_SQLITE3)
  target_link_libraries(dependency_probe PRIVATE SQLite3::SQLite3)
endif ()
